```python
def pick_drills(age, players, space, focus, budget_minutes):
    """按年龄/人数/场地/主题挑训练，塞满给定的分钟预算。"""
    cands = [d for d in DRILLS
             if focus in d["focus"] and age in d["ages"]
             and d["min_players"] <= players <= d["max_players"]
             and space in d["space"]]
    picked, total = [], 0
    for d in cands:
        if total + d["minutes"] <= budget_minutes:
            picked.append(d)
            total += d["minutes"]
    return picked, total


def build_plan(age, players, space, focus, duration):
    if age not in AGES:
        raise ValueError("年龄组必须是 %s 之一" % ", ".join(AGES))
    if space not in SPACES:
        raise ValueError("场地必须是 %s 之一" % ", ".join(SPACES))
    if focus not in FOCUSES:
        raise ValueError("主题必须是 %s 之一" % ", ".join(FOCUSES))
    if players < 4 or players > 22:
        raise ValueError("人数须在 4~22 人之间")
    if duration < 30 or duration > 120:
        raise ValueError("时长须在 30~120 分钟之间")

    warm = max(5, round(duration * 0.10 / 5) * 5)
    cool = 5
    game = 20 if players >= 10 and duration >= 60 else 0
    drill_budget = duration - warm - cool - game

    drills, used = pick_drills(age, players, space, focus, drill_budget)
    if not drills:
        # 主题挑不到时放宽到任意主题（保证总有教案可出）
        drills, used = pick_drills(age, players, space, "passing", drill_budget)
    sections = [{"name": WARMUP["name"], "minutes": warm,
                 "setup": WARMUP["setup"], "points": WARMUP["points"]}]
    for d in drills:
        sections.append({"name": d["name"], "minutes": d["minutes"],
                         "setup": d["setup"], "points": d["points"]})
    if game:
        g = [d for d in DRILLS if d["id"] == "small-sided-5v5"][0]
        sections.append({"name": g["name"], "minutes": game,
                         "setup": g["setup"], "points": g["points"]})
    sections.append({"name": COOLDOWN["name"], "minutes": cool,
                     "setup": COOLDOWN["setup"], "points": COOLDOWN["points"]})
    total = sum(s["minutes"] for s in sections)
    gap = duration - total
    if gap >= 5:
        filler = min(gap, 15)
        sections.insert(len(sections) - 1,
                        {"name": "自由练习与点评（机动）", "minutes": filler,
                         "setup": "按今天主题自由练习，教练逐个点评纠正。",
                         "points": ["只练今天的一个要点", "点评具体到人、到动作"]})
        total = sum(s["minutes"] for s in sections)
    return {"age": age, "players": players, "space": space, "focus": focus,
            "duration_planned": duration, "duration_actual": total,
            "sections": sections}


def render_plan(plan):
    lines = ["# %s · %d 人 · %s场地 · 主题：%s（共 %d 分钟）" % (
        plan["age"], plan["players"],
        {"small": "小", "medium": "中", "full": "全场"}[plan["space"]],
        {"passing": "传球", "dribbling": "带球", "shooting": "射门",
         "defending": "防守", "fitness": "体能"}[plan["focus"]],
        plan["duration_actual"])]
    for i, s in enumerate(plan["sections"], 1):
        lines.append("")
        lines.append("## %d. %s（%d 分钟）" % (i, s["name"], s["minutes"]))
        lines.append("做法：%s" % s["setup"])
        lines.append("要点：" + "；".join(s["points"]))
    return "\n".join(lines)


def project_mrr(subs, arpu, churn, new_subs, months):
    """MRR 推演：每月流失 churn 比例、每月新增 new_subs 人。"""
    rows, s = [], subs
    for m in range(1, months + 1):
        lost = s * churn
        s = s - lost + new_subs
        rows.append({"month": m, "subs": round(s, 1),
                     "mrr": round(s * arpu, 2),
                     "lost": round(lost, 1)})
    return rows

    args = p.parse_args(argv)
    if args.cmd == "plan":
        try:
            plan = build_plan(args.age, args.players, args.space,
                              args.focus, args.duration)
        except ValueError as e:
            print("输入有误：%s" % e, file=sys.stderr)
            return 2
        if args.json:
            print(json.dumps(plan, ensure_ascii=False, indent=2))
        else:
            print(render_plan(plan))
    else:
        rows = project_mrr(args.subs, args.arpu, args.churn,
                           args.new_subs, args.months)
        print("月\t订阅数\tMRR($)\t流失")
        for r in rows:
            print("%d\t%.0f\t%.2f\t%.1f" % (r["month"], r["subs"], r["mrr"], r["lost"]))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

## 📈 模块五：安全合规、防封防踩坑与冷启闭环 SOP

### 红线与避坑

- **教案必须经真教练审校**：AI 生成的教案上线前须经持证教练审一遍；教练对准确度要求极高，一次错误教案就会丢口碑。
- **TrustMRR 口径**：引用收入时以标题口径（MRR $4,778、近 30 天 $4,406）为准，并注明图表口径 $4,202 相差 $204；不得把创始人自述的 19,000+ 注册用户当审计数用。
- **季节性**：青少年足球有赛季周期，淡季收入会掉；MRR 已从 $5,261 降至 $4,778，复制者须每月盯流失率，淡季推室内训练与 FM 战术内容对冲。
- **护城河**：只做纯文字教案会被通用 AI 覆盖；必须守住动画示意图、视频骨架分析、球探数据这三项垂直功能。

### 冷启闭环（首批 10 个付费教练）

1. 找 3 家本地俱乐部：免费整队试用一个月，换口碑与续费。
2. 40+ SEO 页面：按年龄组×主题×场地批量上线，吃长尾搜索。
3. 教练社群真实分享：发教案案例（不发广告），每周 2 篇。
4. 首单验证：免费版用量到限时提醒升级；Pro 周付 $2.99 降低首次付费门槛。
5. 流失看板：每月看 TrustMRR 的 MRR 与流失，流失超 8% 立刻回访差评教练。
