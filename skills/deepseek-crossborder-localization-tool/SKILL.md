## 💻 模块三：生产级 Webhook 与 DeepSeek 极简调度核心源码

### 1. `server.js` (Shopify Webhook 接收与 DeepSeek 翻译中转服务)
```javascript
const express = require("express");
const crypto = require("crypto");
const fetch = require("node-fetch");
const app = express();

app.use(express.json({
  verify: (req, res, buf) => { req.rawBody = buf; }
}));

const SHOPIFY_SECRET = process.env.SHOPIFY_API_SECRET;
const DEEPSEEK_API_KEY = process.env.DEEPSEEK_API_KEY;

// 1. Shopify Webhook 签名 HMAC 安全校验 (防伪造请求)
function verifyShopifyHmac(req) {
  const hmacHeader = req.get("X-Shopify-Hmac-Sha256");
  if (!hmacHeader) return false;
  const hash = crypto
    .createHmac("sha256", SHOPIFY_SECRET)
    .update(req.rawBody, "utf8")
    .digest("base64");
  return crypto.timingSafeEqual(Buffer.from(hash), Buffer.from(hmacHeader));
}

// 2. 核心监听端点：商品创建或更新时自动触发翻译
app.post("/webhooks/products/update", async (req, res) => {
  if (!verifyShopifyHmac(req)) {
    return res.status(401).send("Unauthorized Webhook");
  }

  // 立即返回 200 响应给 Shopify，防止超时重试
  res.status(200).send("Accepted");

  const product = req.body;
  const targetLanguages = ["es", "ar", "de"]; // 卖家配置的目标语种

  for (const lang of targetLanguages) {
    try {
      const translatedTitle = await translateWithDeepSeek(product.title, lang, "title");
      const translatedBodyHtml = await translateWithDeepSeek(product.body_html, lang, "html");

      // 将翻译后的数据写回 Shopify GraphQL Admin API
      await syncTranslationToShopify(product.admin_graphql_api_id, lang, {
        title: translatedTitle,
        body_html: translatedBodyHtml
      });
      console.log(`[Success] Translated product ${product.id} to ${lang}`);
    } catch (err) {
      console.error(`[Error] Failed to translate product ${product.id} to ${lang}:`, err.message);
    }
  }
});

// 3. DeepSeek API 电商特化请求包装
async function translateWithDeepSeek(content, targetLang, contentType = "text") {
  if (!content) return "";

  const systemPrompt = `You are an elite cross-border e-commerce localization expert.
Translate the following product ${contentType} into natural, high-converting ${targetLang}.
CRITICAL RULES:
1. Preserve ALL HTML tags, placeholders, and brand names exactly as they are.
2. Adapt marketing idioms and measurements to local buyer habits.
3. Return ONLY the translated string with no explanations or conversational filler.`;

  const response = await fetch("https://api.deepseek.com/chat/completions", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${DEEPSEEK_API_KEY}`
    },
    body: JSON.stringify({
      model: "deepseek-chat",
      messages: [
        { role: "system", content: systemPrompt },
        { role: "user", content: content }
      ],
      temperature: 0.2
    })
  });

  const data = await response.json();
  return data.choices[0].message.content.trim();
}

app.listen(3000, () => console.log("DeepSeek Localization Service running on port 3000"));
```

## 🎯 模块四：电商转化特化型 Prompt 编排链 (反生硬机翻)

在处理商品详情页时，通用机翻最大的毛病是破坏 HTML 结构或把核心品牌名词强行直译。请在生产中锁死以下系统级 Prompt：

```text
# Role: 跨境电商高级本地化翻译引擎 (E-Commerce Cultural Localization Master)

## 任务目标:
将输入的独立站商品文案精准转化为目标语言（${target_lang}）。不仅要做到词义准确，更要达到当地头部 DTC 独立站的高转化营销语感。

## 严格执行铁律:
1. 【HTML 结构与短代码绝对锁死】：所有的 <div>, <p>, <span>, <strong>, 以及 Shopify 短代码（如 {{product.price}}）原样保留，严禁闭合错位或丢失标签；
2. 【专有名词保护】：品牌商标、认证型号（如 CE, FCC, ISO）、特定材质专有名词不得擅自翻译；
3. 【结账心理学与行动召唤 (CTA) 本地化】：将生硬的直译（如 "立即购买"）转化为当地消费者最具冲动感的高频习惯短语（如西语: "¡Compra Ahora con Envío Gratis!"；德语: "Jetzt sicher bestellen"）；
4. 【纯净输出契约】：输出内容直接替换原文，禁止输出任何前缀、解释、双引号或 Markdown 代码块包裹。
```
