### 2. `license_verify.js` (LemonSqueezy 离线防盗与缓存校验核心逻辑)
```javascript
// LemonSqueezy Merchant of Record (MoR) License Verification Module
const LEMON_API_URL = "https://api.lemonsqueezy.com/v1/licenses/activate";

export async function verifyLicense(licenseKey, instanceName = "Chrome-Client") {
  if (!licenseKey || licenseKey.trim().length < 10) {
    return { valid: false, error: "Invalid license key format" };
  }

  // 1. 优先读取本地缓存 (支持 7 天免联网离线可用，极大提升体验)
  const cached = await chrome.storage.local.get(["license_cache", "license_last_check"]);
  const now = Date.now();
  const SEVEN_DAYS = 7 * 24 * 60 * 60 * 1000;

  if (cached.license_cache === licenseKey && cached.license_last_check && (now - cached.license_last_check < SEVEN_DAYS)) {
    return { valid: true, cached: true };
  }

  // 2. 联网向 LemonSqueezy 发起激活与校验
  try {
    const response = await fetch(LEMON_API_URL, {
      method: "POST",
      headers: {
        "Accept": "application/json",
        "Content-Type": "application/x-www-form-urlencoded"
      },
      body: new URLSearchParams({
        license_key: licenseKey.trim(),
        instance_name: instanceName
      })
    });

    const data = await response.json();
    if (data.activated && data.license_key && data.license_key.status === "active") {
      // 校验成功，写入本地隔离存储
      await chrome.storage.local.set({
        license_cache: licenseKey.trim(),
        license_status: "pro",
        license_last_check: now
      });
      return { valid: true, meta: data.meta };
    } else {
      await chrome.storage.local.set({ license_status: "free" });
      return { valid: false, error: data.error || "License not active or seat limit reached" };
    }
  } catch (err) {
    // 若网络临时波动且本地曾验证通过，降级宽限放行
    if (cached.license_cache === licenseKey) {
      return { valid: true, grace_period: true };
    }
    return { valid: false, error: "Network error during license verification" };
  }
}
```
