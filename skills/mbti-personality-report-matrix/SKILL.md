def _extract_sc_otf(ttc_path, cache_dir):
    """从 TTC 合集里抽出简体中文（SC）子字体，存为独立 .otf；返回路径。
    单语种 TTC（如微软雅黑）没有 SC 标记时，直接取第一个子字体。"""
    os.makedirs(cache_dir, exist_ok=True)
    base = os.path.splitext(os.path.basename(ttc_path))[0]
    out_path = os.path.join(cache_dir, base + "-SC.otf")
    if os.path.exists(out_path):
        return out_path
    coll = TTCollection(ttc_path)
    target = None
    for font in coll.fonts:
        name = font["name"].getDebugName(1) or ""
        if "SC" in name or "CN" in name:
            target = font
            break
    if target is None:
        target = coll.fonts[0]
    target.save(out_path)
    return out_path


def resolve_fonts():
    """返回 (regular_otf, bold_otf)；找不到则明确报错，不静默出乱码。"""
    cache_dir = os.path.join(os.path.expanduser("~"), ".cache", "mbti-report")
    regular = bold = None
    for path in FONT_CANDIDATES:
        if os.path.exists(path):
            try:
                regular = _extract_sc_otf(path, cache_dir)
                break
            except Exception:
                continue
    for path in BOLD_CANDIDATES:
        if os.path.exists(path):
            try:
                bold = _extract_sc_otf(path, cache_dir)
                break
            except Exception:
                continue
    if regular is None:
        raise RuntimeError("未找到可用的中文字体：请安装 Noto CJK 字体，"
                           "或把字体路径加进 FONT_CANDIDATES。")
    return regular, bold or regular

class ReportPDF(FPDF):
    """带页脚（合规声明 + 页码）的报告 PDF。"""

    def footer(self):
        self.set_y(-15)
        self.set_font("CJK", "", 8)
        self.set_text_color(136, 136, 136)
        self.cell(0, 10, "MBTI 人格解析报告 · 趣味测试，仅供自我探索参考，不构成心理诊断"
                 "  第 %d 页" % self.page_no(), align="C")


def build_report(mbti, theme, out_dir):
    """生成单份 PDF 报告；返回输出路径。"""
    mbti = mbti.upper()
    if mbti not in TYPE_BANK:
        raise ValueError("未知类型：%s（应为 16 型之一，如 INTJ）" % mbti)
    if theme not in THEMES:
        raise ValueError("未知主题：%s（可选：%s）" % (theme, "、".join(THEMES)))
    regular_otf, bold_otf = resolve_fonts()
    info = TYPE_BANK[mbti]
    price, strike = PRICE_LADDER[theme]
    os.makedirs(out_dir, exist_ok=True)
    out_path = os.path.join(out_dir, "%s-%s解析报告.pdf" % (mbti, theme))

    pdf = ReportPDF("P", "mm", "A4")
    pdf.set_auto_page_break(auto=True, margin=20)
    pdf.set_margins(20, 18, 20)
    pdf.add_font("CJK", "", regular_otf)
    pdf.add_font("CJK", "B", bold_otf)
    pdf.set_title("%s %s解析报告" % (mbti, theme))
    pdf.set_author("MBTI 人格解析报告")

    # 封面
    pdf.add_page()
    pdf.ln(28)
    pdf.set_font("CJK", "B", 36)
    pdf.set_text_color(26, 26, 46)
    pdf.cell(0, 16, mbti, new_x="LMARGIN", new_y="NEXT", align="C")
    pdf.ln(4)
    pdf.set_font("CJK", "", 15)
    pdf.set_text_color(85, 85, 85)
    pdf.cell(0, 10, "%s · %s深度解析" % (info["name"], theme),
            new_x="LMARGIN", new_y="NEXT", align="C")

    pdf.cell(0, 10, info["tagline"], new_x="LMARGIN", new_y="NEXT", align="C")

# ------------------------------------------------------------- SKU 矩阵
def build_matrix(out_dir):
    """生成 16 型 × 4 主题的 SKU 矩阵 CSV（64 行）；返回输出路径。"""
    os.makedirs(out_dir, exist_ok=True)
    out_path = os.path.join(out_dir, "mbti-sku-matrix.csv")
    with open(out_path, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.writer(f)
        writer.writerow(["sku_id", "mbti", "type_name", "theme", "title",
                         "price", "strike_price", "role"])
        for mbti, info in TYPE_BANK.items():
            for theme in THEMES:
                price, strike = PRICE_LADDER[theme]
                role = "引流款" if theme == "七宗罪分布" else (
                    "主力款" if theme == "心智阶段" else "利润款")
                writer.writerow([
                    "mbti-%s-%s" % (mbti.lower(), THEMES.index(theme) + 1),
                    mbti, info["name"], theme,
                    "%s %s%s解析报告" % (mbti, info["name"], theme),
                    price, strike, role,
                ])
    return out_path


# ---------------------------------------------------------------- 入口
def main(argv=None):
    parser = argparse.ArgumentParser(description="MBTI 人格解析报告 PDF 批量生成器")
    parser.add_argument("--type", default="INTJ", help="人格类型，如 INTJ（默认 INTJ）")
    parser.add_argument("--theme", default="心智阶段",
                        help="报告主题：心智阶段/恋爱风格/职业适配/七宗罪分布")
    parser.add_argument("--all", action="store_true", help="生成全部 16 型的心智阶段报告")
    parser.add_argument("--matrix", action="store_true", help="生成 SKU 矩阵 CSV")
    parser.add_argument("--out", default="./out", help="输出目录（默认 ./out）")
    args = parser.parse_args(argv)

    if args.matrix:
        path = build_matrix(args.out)
        print("SKU 矩阵已生成：%s（64 个 SKU）" % path)
    if args.all:
        made = [build_report(t, args.theme, args.out) for t in TYPE_BANK]
        print("已生成 %d 份报告，目录：%s" % (len(made), args.out))
    elif not args.matrix:
        path = build_report(args.type, args.theme, args.out)
        print("报告已生成：%s" % path)
    return 0


if __name__ == "__main__":
    sys.exit(main())
