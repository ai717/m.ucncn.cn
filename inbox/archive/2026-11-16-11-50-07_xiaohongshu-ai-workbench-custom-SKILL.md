## 💻 模块三：生产级极简核心工程源码 / 数据库架构

本模块提供真实、可用、可直接复制运行的完整核心代码 `workbench_quote.py`（意向评分 + 报价 + 需求锁定合同），已实测通过（实测记录见 README）。严禁写伪代码或占位符。

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""小红书获客接 AI 工作台定制 · 报价与需求锁定脚手架

用途：把复盘文章里的真实打法变成可跑的工具——
  1) 客户意向评分（漏斗筛选）：按文章里的 6 个筛选问题给来访者打分；
  2) 报价单生成：按真实锚点价格（¥500 测试 / ¥1,600 / ¥2,800 / ¥3,660 / 增功能 ¥800）出报价档位；
  3) 需求锁定合同：第一次报价只对应确认的范围，新增功能再谈，防无限加需求。

真实锚点（2026-10-26 向前精选复盘原文，C 级自述，未审计）：
  测试 ¥500 / 个人财务工作台 ¥1,600 / 中药材系统 ¥2,800（后增功能另报 ¥800）
  / 牙科外贸工作台首版 ¥3,660 / 功能增加后累计 ¥8,900。
本工具不虚构价格：报价只输出锚点档位，档位之间不插值编造。

用法：
  python3 workbench_quote.py --demo     # 跑三个真实案例 + 意向评分演示
  python3 workbench_quote.py --check    # 自测：断言三个真实案例的报价与锚点一致
  python3 workbench_quote.py --quote '{"industry": "中药材", "links": ["采购", "加工", "库存", "销售", "回款"], "multi_user": true, "permissions": true, "deploy": true, "after_sales": true}'
"""

import json
import sys

# ---- 真实锚点价格（C 级自述，见复盘原文成绩板与三个实际案例）----
TEST_FEE = 500
ANCHOR_SMALL = 1600        # 个人财务工作台：记账 / 发票 / 财务信息整理（3 个环节）
ANCHOR_MEDIUM = 2800       # 中药材系统：采购 / 加工 / 库存 / 销售 / 回款（5 个环节）
ANCHOR_MEDIUM_PLUS = 3660  # 牙科外贸工作台首版：接病历图 / 录单 / 分配供应商 / 跟进度 / 生成账单收款（6 个环节）
ANCHOR_EXTRA_FEATURE = 800  # 中药材后增功能另报 800 元

# ---- 漏斗筛选：6 个问题（复盘原文“筛选先聊清”）----
LEAD_QUESTIONS = [
    ("frequency", "这件事多久发生一次？", 15),
    ("current_way", "现在怎么处理？", 15),
    ("pain_point", "哪里费时间、易出错？", 15),
    ("users", "谁用？", 10),
    ("decision_maker", "谁拍板？", 15),
    ("budget", "有没有预算？", 20),
    ("timeline", "什么时候要用？", 10),
]


def screen_lead(answers):
    """给一个来访者打意向分。answers: {问题 key: 是否答清楚了 True/False}。
    返回 (分数 0-100, 结论）。"""
    score = sum(w for k, _, w in LEAD_QUESTIONS if answers.get(k))
    if score >= 80:
        verdict = "高意向：约需求访谈，让客户拿一笔真实业务从头讲一遍"
    elif score >= 50:
        verdict = "中等意向：继续追问，没答清的问题逐个补"
    else:
        verdict = "低意向：先放资料池，别急着报价"
    return score, verdict


def quote_project(links, multi_user=False, permissions=False, deploy=False, after_sales=False):
    """按业务环节数出报价档位。links: 客户确认的业务环节清单。
    返回 dict：档位、报价、测试费、交付清单、合同要点。
    报价只输出真实锚点档位，不插值编造。"""
    n = len(links)
    if n <= 3:
        price, tier = ANCHOR_SMALL, "小型工作台（1-3 个业务环节）"
    elif n <= 5:
        price, tier = ANCHOR_MEDIUM, "中型系统（4-5 个业务环节）"
    else:
        price, tier = ANCHOR_MEDIUM_PLUS, "中大型工作台（6 个以上业务环节）"
    delivery = ["需求确认与流程图", "测试版本交付与验收"]
    if multi_user:
        delivery.append("多人操作：账号与数据权限划分")
    if permissions:
        delivery.append("权限：谁能看、谁能改、谁能导出")
    if deploy:
        delivery.append("部署：服务器 / 域名 / 模型接口费用提前聊")
    if after_sales:
        delivery.append("售后：响应时效与免费维护期写进合同")
    return {
        "tier": tier,
        "links": links,
        "test_fee": TEST_FEE,
        "quote": price,
        "extra_feature_reference": ANCHOR_EXTRA_FEATURE,
        "delivery_checklist": delivery,
        "scope_rule": "第一次报价只对应确认的范围，新增功能再谈（参考价 ¥800/次）",
    }


def scope_lock_contract(client, industry, links, quote):
    """生成需求锁定合同（Markdown）。只锁确认的范围，其余一律另报价。"""
    lines = [
        "# 工作台定制 · 需求锁定确认单",
        "",
        f"客户：{client}（{industry}）",
        f"档位：{quote['tier']}，报价：¥{quote['quote']}（测试费 ¥{quote['test_fee']} 另计，先收测试费再做正式项目）",
        "",
        "## 确认的业务范围（只做这些）",
        "",
    ]
    for link in links:
        lines.append(f"- {link}")
    lines += [
        "",
        "## 不包含（需要再谈、另报价）",
        "",
        "- 上面没列的功能，一律不包含在本次报价内",
        "- 增功能参考价 ¥800/次，具体按技术伙伴评估另报",
        "",
        "## 交付清单",
        "",
    ]
    for item in quote["delivery_checklist"]:
        lines.append(f"- [ ] {item}")
    lines += [
        "",
        "## 边界条款",
        "",
        "- 第一次报价只对应确认的范围，新增功能再谈",
        "- 技术评估前不把话说满：工期与功能实现以技术伙伴评估为准",
        "- 测试版本、正式开发、新增功能分清，先做测试再做正式项目",
        "- 服务器、域名、模型接口等费用提前聊清，不事后加价",
        "",
    ]
    return "\n".join(lines)


def demo():
    """跑三个真实案例 + 一次意向评分演示。"""
    cases = [
        ("个人财务工作台", ["记账", "发票整理", "财务信息整理"], {}),
        ("中药材系统", ["采购", "加工", "库存", "销售", "回款"],
         {"multi_user": True, "permissions": True, "deploy": True, "after_sales": True}),
        ("牙科外贸工作台首版", ["接收病历图片", "录单", "分配供应商", "跟进度", "生成账单", "收款"],
         {"multi_user": True, "permissions": True, "deploy": True, "after_sales": True}),
    ]
    out = []
    for name, links, flags in cases:
        q = quote_project(links, **flags)
        out.append({"case": name, "quote": q["quote"], "tier": q["tier"]})
        print(f"【{name}】{q['tier']}，报价 ¥{q['quote']}，测试费 ¥{q['test_fee']}")
    lead = {"frequency": True, "current_way": True, "pain_point": True,
            "users": True, "decision_maker": True, "budget": True, "timeline": False}
    score, verdict = screen_lead(lead)
    print(f"【意向评分演示】{score} 分：{verdict}")
    out.append({"lead_demo": {"score": score, "verdict": verdict}})
    print(json.dumps(out, ensure_ascii=False, indent=2))
    return out


def check():
    """自测：三个真实案例的报价必须与原文锚点一致；意向评分阈值必须正确。"""
    assert quote_project(["记账", "发票整理", "财务信息整理"])["quote"] == 1600
    assert quote_project(["采购", "加工", "库存", "销售", "回款"])["quote"] == 2800
    assert quote_project(["接收病历图片", "录单", "分配供应商", "跟进度", "生成账单", "收款"])["quote"] == 3660
    assert quote_project(["记账"])["quote"] == 1600  # 1-3 环节同档
    assert quote_project(["采购", "加工", "库存", "销售", "回款", "财务"])["quote"] == 3660  # 6 环节进高档
    s, _ = screen_lead({k: True for k, _, _ in LEAD_QUESTIONS})
    assert s == 100
    s, v = screen_lead({"budget": True})
    assert s == 20 and "低意向" in v
    s, v = screen_lead({"frequency": True, "current_way": True, "pain_point": True,
                        "users": True, "decision_maker": True, "budget": True})
    assert s == 90 and "高意向" in v
    contract = scope_lock_contract("测试客户", "中药材", ["采购", "加工", "库存", "销售", "回款"],
                                   quote_project(["采购", "加工", "库存", "销售", "回款"]))
    assert "¥2800" in contract and "新增功能再谈" in contract and "多人操作" not in contract
    contract2 = scope_lock_contract("测试客户", "牙科外贸",
                                    ["接收病历图片", "录单", "分配供应商", "跟进度", "生成账单", "收款"],
                                    quote_project(["接收病历图片", "录单", "分配供应商", "跟进度", "生成账单", "收款"],
                                                  multi_user=True, permissions=True, deploy=True, after_sales=True))
    assert "多人操作" in contract2 and "部署" in contract2 and "¥3660" in contract2
    print("全部自测通过：3 个真实案例报价与原文锚点一致，意向评分与合同生成正常。")


if __name__ == "__main__":
    if "--check" in sys.argv:
        check()
    elif "--quote" in sys.argv:
        i = sys.argv.index("--quote")
        data = json.loads(sys.argv[i + 1])
        q = quote_project(data["links"], multi_user=data.get("multi_user", False),
                          permissions=data.get("permissions", False),
                          deploy=data.get("deploy", False),
                          after_sales=data.get("after_sales", False))
        print(json.dumps(q, ensure_ascii=False, indent=2))
        print(scope_lock_contract(data.get("client", "客户"), data.get("industry", ""),
                                  data["links"], q))
    else:
        demo()
```

## 🎯 模块四：行业特化型 Prompt 编排链（需求问诊 + 获客内容 + 报价沟通）

### 4.1 需求问诊 Prompt（核心资产：让客户讲一笔真实业务）

```
你正在帮一位传统小商家（中药材 / 牙科外贸 / 个体财务）梳理他到底要什么工作台。
禁止做的事：不要只记“库存管理”“自动出账”这种功能名；不要在技术评估前把话说满；不要问“你要什么功能”。
必须做的事：让他拿一笔刚发生过的业务从头讲一遍，顺着追问——
1. 这笔订单怎么来的？
2. 收到先做什么、记在哪、下一步交给谁、什么时候算完成？
3. 这件事多久发生一次？现在怎么处理？哪里费时间、容易出错？
4. 谁用？谁拍板？有没有预算？什么时候要用？
5. 先收集客户现有表格、截图、文件，整理流程，确认重复录入和易漏点，再出方案。
6. 客户确认方案后，再和技术伙伴评估实现范围和报价；第一次报价只对应确认的范围，新增功能再谈。
输出：一份“业务流程记录 + 确认的业务环节清单 + 不包含项”，交给技术伙伴评估。
```

### 4.2 小红书获客内容 Prompt

```
你是小红书内容操盘手，目标是把“有定制需求的传统小商家”筛出来，不是涨粉。
内容方向只做三类：资料（可下载的 ¥9.9 资料/教程）、教程（一步步教他处理某件具体事）、工作台内容（展示一个工作台怎么帮他处理订单/库存/回款）。
禁止：直接发“AI 智能体定制”“企业工作流搭建”这类离用户远的概念。
每篇内容必须回答一个问题：“这件事能帮我处理什么？”（用户每天也在做的那件具体事）。
对标判断：不只看流量，看是否有人买单、多篇有反馈、评论区有没有反复出现的问题。
```

### 4.3 报价沟通话术（第一次报价模板）

```
报价只对应确认的范围，新增功能再谈。话术骨架：
- “测试费 ¥500，先做一个小版本验证，跑通了再做正式项目。”
- “正式项目报价 ¥X，只包括我们刚才确认的这几个环节：……”
- “上面没列的功能不在本次报价内，需要的话我们再评估另报（参考 ¥800/次）。”
- “工期和实现方式以技术伙伴评估为准，我不先把话说满。”
- “服务器、域名、模型接口这些费用我们现在就聊清，不事后加价。”
```

## 📈 模块五：安全合规、防封防踩坑与冷启闭环 SOP

### 5.1 三条红线（写进合同，缺一不可）
1. **技术伙伴前置**：没有靠谱技术伙伴不接单。作者管内容获客、沟通、需求整理、成交推进；技术伙伴管评估、开发、交付。技术上的工期和功能实现不能为了促成合作自己先答应。
2. **需求锁定**：第一次报价只对应确认的范围；测试版本、正式开发、新增功能分清；不把范围锁死会被无限加需求拖垮。
3. **交付清单**：给客户用的项目必须过清单——多人操作、数据保存、权限、部署、售后；服务器、域名、模型接口等费用提前聊。

### 5.2 冷启 14 天节拍器
- Day 1~3：做 ¥9.9 资料/教程，发 3 篇获客内容，测“工作台”相关搜索词有没有真实流量。
- Day 4~7：进群筛选，用 6 个问题筛意向；每天新增里大概五六个会明确咨询。
- Day 8~10：约高意向客户做需求访谈，让客户讲一笔真实业务；整理流程给技术伙伴评估。
- Day 11~12：先收 ¥500 测试费做小版本，跑通再谈正式项目报价。
- Day 13~14：交付测试版，拿客户反馈发朋友圈/群，启动转介绍。

### 5.3 常见翻车点
- 听到功能名就以为听懂了 → 必须让客户讲一笔真实业务。
- 技术评估前把话说满 → 工期与功能实现以技术伙伴评估为准。
- 范围没锁死 → 增功能必须另报价，写进合同。
- 只看流量不看买单 → 对标看有没有人真掏钱、评论区有没有真实需求。
- 数字当审计数宣传 → 全部收入数字均为作者自述，无收款截图佐证；对外只讲“复盘文章自述”，不承诺收益。
