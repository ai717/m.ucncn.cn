## 💻 模块三：生产级极简核心工程源码 / 数据库架构

> 【硬性门禁】：以下代码真实、可用、可直接复制运行，严禁写伪代码或占位符。已实测：24 页涂色书计划生成、角色锚点逐页一致、两种 KDP 规格验算、漏斗示例、页数下限报错均通过。

```python
#!/usr/bin/env python3
"""ColorCraft KDP 出书工具脚手架：出书计划 + KDP 印刷规格 + 试用转化漏斗。
零依赖，只用 Python 标准库。用法：
  python3 colorcraft_kdp.py plan --type coloring --theme 森林动物 --pages 24
  python3 colorcraft_kdp.py specs --trim 8.5x11 --pages 24 --paper white
  python3 colorcraft_kdp.py funnel --demo
"""
import argparse
import csv
import json
import sys

BLEED_IN = 0.125          # KDP 出血：每边 0.125 英寸
MIN_PAGES = 24            # KDP 平装最低页数（不含封面）
MAX_PAGES = 828           # KDP 平装最高页数
PAPER_THICKNESS = {"white": 0.002252, "cream": 0.0025}  # 英寸/页，Amazon 官方口径
DPI = 300                 # KDP 要求内页图片 300 DPI

BOOK_TYPES = {
    "coloring": {
        "label": "涂色书",
        "page_kinds": ["封面", "版权页", "涂色页", "涂色页", "涂色页", "涂色页", "结尾页"],
        "prompt_tpl": "black and white line art coloring page, clean bold outlines, white background, no shading, {subject}, consistent character: {anchors}",
    },
    "story": {
        "label": "故事书",
        "page_kinds": ["封面", "版权页", "故事页", "故事页", "故事页", "故事页", "结尾页"],
        "prompt_tpl": "children's picture book illustration, warm watercolor style, {subject}, consistent character: {anchors}",
    },
    "comic": {
        "label": "漫画书",
        "page_kinds": ["封面", "版权页", "分镜页", "分镜页", "分镜页", "分镜页", "结尾页"],
        "prompt_tpl": "comic book panel, lettered, clean ink lines, {subject}, consistent character: {anchors}",
    },
    "puzzle": {
        "label": "解谜书",
        "page_kinds": ["封面", "版权页", "谜题页", "谜题页", "谜题页", "谜题页", "答案页"],
        "prompt_tpl": "printable puzzle page for kids, clear black-and-white layout, {subject}, consistent character: {anchors}",
    },
}

# 主角一致性锚点：每页提示词原文复用同一段描述，角色才不会"换脸"。
CHARACTER_ANCHORS = {
    "森林动物": "a small orange fox named Doudou, white belly, green scarf, round ears",
    "太空冒险": "a young astronaut girl named Luna, silver suit, red helmet stripe, curly black hair",
    "海底世界": "a blue baby whale named Bubbles, white belly, tiny yellow crown",
    "恐龙乐园": "a green baby T-Rex named Rocky, tiny arms, red sneakers",
}


def character_anchors(theme):
    return CHARACTER_ANCHORS.get(theme, "a friendly animal hero, same outfit and colors on every page")


def build_plan(book_type, theme, pages):
    """生成整书页计划：每页一种页型 + 一条带角色锚点的生图提示词。"""
    if book_type not in BOOK_TYPES:
        raise ValueError("book_type 必须是 %s 之一" % "/".join(BOOK_TYPES))
    if not MIN_PAGES <= pages <= MAX_PAGES:
        raise ValueError("KDP 平装页数须在 %d~%d 之间，当前 %d" % (MIN_PAGES, MAX_PAGES, pages))
    spec = BOOK_TYPES[book_type]
    anchors = character_anchors(theme)
    kinds = spec["page_kinds"]
    plan = []
    for i in range(1, pages + 1):
        kind = kinds[0] if i == 1 else kinds[1] if i == 2 else kinds[-1] if i == pages else kinds[2 + (i % 4)]
        subject = "%s，主题 %s，第 %d 页" % (spec["label"], theme, i)
        prompt = spec["prompt_tpl"].format(subject=subject, anchors=anchors)
        plan.append({"page": i, "kind": kind, "prompt": prompt})
    return {
        "book_type": book_type,
        "label": spec["label"],
        "theme": theme,
        "pages": pages,
        "character_anchors": anchors,
        "plan": plan,
    }


def kdp_specs(trim, pages, paper):
    """算 KDP 印刷规格：书脊宽、封面尺寸、内页像素。trim 如 8.5x11（英寸）。"""
    try:
        w, h = (float(x) for x in trim.lower().split("x"))
    except ValueError:
        raise ValueError("trim 格式应为 宽x高，如 8.5x11")
    if paper not in PAPER_THICKNESS:
        raise ValueError("paper 必须是 white 或 cream")
    if not MIN_PAGES <= pages <= MAX_PAGES:
        raise ValueError("KDP 平装页数须在 %d~%d 之间，当前 %d" % (MIN_PAGES, MAX_PAGES, pages))
    spine = round(pages * PAPER_THICKNESS[paper], 4)
    cover_w = round(w * 2 + spine + BLEED_IN * 2, 3)
    cover_h = round(h + BLEED_IN * 2, 3)
    return {
        "trim_in": "%sx%s" % (w, h),
        "pages": pages,
        "paper": paper,
        "spine_in": spine,
        "cover_in": "%s x %s" % (cover_w, cover_h),
        "interior_px_at_300dpi": "%d x %d" % (int(w * DPI), int(h * DPI)),
        "bleed_in": BLEED_IN,
        "checks": [
            "页数 %d 在 KDP 平装区间 %d~%d 内" % (pages, MIN_PAGES, MAX_PAGES),
            "封面含出血，每边 +0.125 英寸",
            "内页图片按 300 DPI 备图",
            "上传前须在 KDP 说明 AI 生成内容（KDP 要求披露）",
        ],
    }


def funnel_report(rows):
    """试用转化漏斗：rows 为 (trial_started, books_completed, paid) 三元组。
    核心指标：完成整本才付费——ColorCraft 的发现是 10 页以上用户才付费。"""
    started = sum(r[0] for r in rows)
    completed = sum(r[1] for r in rows)
    paid = sum(r[2] for r in rows)
    if started == 0:
        raise ValueError("试用开始数为 0，无法计算转化率")
    return {
        "trial_started": started,
        "books_completed": completed,
        "paid": paid,
        "completion_rate": round(completed / started * 100, 1),
        "paid_per_completed_pct": round(paid / completed * 100, 1) if completed else 0.0,
        "overall_conversion_pct": round(paid / started * 100, 1),
        "verdict": "完成整本的用户付费率 %.1f%%，做完一整本才收费的 7 天试用是关键设计" % (
            paid / completed * 100 if completed else 0.0),
    }


def main(argv=None):
    ap = argparse.ArgumentParser(description="ColorCraft KDP 出书工具脚手架")
    sub = ap.add_subparsers(dest="cmd", required=True)
    p = sub.add_parser("plan", help="生成整书页计划")
    p.add_argument("--type", choices=list(BOOK_TYPES), required=True)
    p.add_argument("--theme", required=True)
    p.add_argument("--pages", type=int, default=24)
    p.add_argument("--out", help="写出 JSON 文件，不填则打印")
    s = sub.add_parser("specs", help="算 KDP 印刷规格")
    s.add_argument("--trim", default="8.5x11")
    s.add_argument("--pages", type=int, default=24)
    s.add_argument("--paper", choices=list(PAPER_THICKNESS), default="white")
    f = sub.add_parser("funnel", help="试用转化漏斗（--demo 用示例数据）")
    f.add_argument("--demo", action="store_true")
    f.add_argument("--csv", help="CSV 文件：trial_started,books_completed,paid 三列")
    args = ap.parse_args(argv)
    if args.cmd == "plan":
        data = build_plan(args.type, args.theme, args.pages)
        text = json.dumps(data, ensure_ascii=False, indent=2)
        if args.out:
            open(args.out, "w", encoding="utf-8").write(text)
            print("已写出 %s（%d 页）" % (args.out, args.pages))
        else:
            print(text)
    elif args.cmd == "specs":
        print(json.dumps(kdp_specs(args.trim, args.pages, args.paper), ensure_ascii=False, indent=2))
    elif args.cmd == "funnel":
        if args.demo:
            rows = [(100, 42, 18), (120, 55, 24), (90, 38, 15)]
        elif args.csv:
            rows = [(int(r[0]), int(r[1]), int(r[2])) for r in csv.reader(open(args.csv, encoding="utf-8")) if r]
        else:
            ap.error("funnel 需要 --demo 或 --csv")
        print(json.dumps(funnel_report(rows), ensure_ascii=False, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### 实测记录（2026-11-16）

- `plan --type coloring --theme 森林动物 --pages 24`：出 24 页计划，主角锚点逐页一致（24/24 页含同一锚点）。

- **解谜书**：`printable puzzle page for kids, clear black-and-white layout, [画面], consistent character: [锚点]`

### 3. KDP 上架文案模板（标题 + 7 个关键词）

- 标题：`[主题] Coloring Book for Kids Ages 4-8: [页数] Fun [主题] Coloring Pages`
- 副标题：`A delightful [主题] adventure coloring book with consistent characters and print-ready pages`
- 关键词（7 个，填 KDP 后台）：主题词 2 个 + 人群词 2 个 + 场景词 2 个 + 格式词 1 个（例：forest animals, kids coloring book, ages 4-8, animal adventure, printable, gift for kids, KDP paperback）

## 📈 模块五：安全合规、防封防踩坑与冷启闭环 SOP

### 合规红线

- **KDP 披露义务**：上传时必须说明用了 AI 生成内容；低质量 AI 书会被限制展示，别拿模板批量刷书，一本一本好好做。
- **版权**：别用现成 IP 角色（迪士尼、宝可梦等）当主角；角色锚点用自己编的形象。
- **收入宣传**：引用运营者数字时注明来源（TrustMRR/Stripe，2026-11-15 更新）与核查方式；别把 $2,388 MRR 写成"保证收入"。
- **图片规格**：内页 300 DPI、出血每边 0.125 英寸、页数 24~828，上传前用 KDP 预览器逐页检查。

### 冷启闭环（前 10~50 个种子付费用户）

1. **第 1 天**：跑通脚手架，做出第一本 24 页涂色书并上架 KDP，全程截图留作过程贴素材。
2. **第 2~3 天**：在 KDP 社区、Reddit 自出版板块、Facebook 自出版小组发过程贴（"我用一句话出了一本涂色书，全流程"），别发硬广，附试用链接。
3. **第 1~2 周**：写 10 篇长尾内容（"涂色书 KDP 上架全流程""KDP 封面尺寸怎么算"），拿搜索流量；同步记漏斗 CSV，每周看一次试用→做完→付费。
4. **目标**：50 个试用用户；做完率低于 30% 先改试用引导，别急着加功能。
