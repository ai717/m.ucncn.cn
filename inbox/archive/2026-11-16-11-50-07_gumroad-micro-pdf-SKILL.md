## 💻 模块三：生产级极简核心工程源码 / 数据库架构

脚手架 `micro_pdf.py`：痛点打分 + PDF 生成 + 上架文案包 + 销售台账，四合一单文件（仅依赖 fpdf2）。**已实测通过**（2026-11-16，见模块五末尾的实测记录）：

- 痛点打分三例：咖啡机除味 81.5（做）、贴纸残胶 68.5（可做）、线缆收纳 56.5（别做）；
- PDF 生成：咖啡机指南 1 页，逐份提取文字校验标题、步骤、贴士、检查清单、价格齐全；
- 上架文案包：标题/描述/定价/3 条 BTW 模板 + 3 条防垃圾信息规则；
- 销售台账：2 个产品、20 单、$91.80 汇总无误。
- 修过一个真问题：fpdf2 核心字体只支持 latin-1，em-dash 会炸掉整个生成——已加 `_ascii()` 统一折叠 Unicode 标点。

```python
#!/usr/bin/env python3
"""micro_pdf.py -- Gumroad 微型数字产品（小 PDF 变现）脚手架。

原型 D：一次制作、零边际成本的数字化资产。功能：
  score    给候选小麻烦打分（出现频率 x 烦人程度 x 付费意愿 x 可搜索性）
  build    把结构化指南 spec（JSON）渲染成可直接上架 Gumroad 的 PDF
  listing  生成 Gumroad 上架文案包（标题/描述/定价/“BTW” 留言模板）
  catalog  销售台账：登记产品与销量，汇总总收入（对标 Gloria 30 天 $1,171.92）

用法：
  python micro_pdf.py score --problem spec.json
  python micro_pdf.py build --spec spec.json --out guide.pdf
  python micro_pdf.py listing --spec spec.json
  python micro_pdf.py catalog add --file catalog.json --slug coffee-maker-fix --price 4.99 --sales 12
  python micro_pdf.py catalog total --file catalog.json
"""
import argparse
import json
import os
import sys

# ---------------------------------------------------------------- score
# 权重：付费意愿最重（没人掏钱 = 白忙），其次是可搜索性（决定能不能被找到）
WEIGHTS = {"frequency": 0.25, "annoyance": 0.25, "willingness": 0.30, "searchability": 0.20}


def score_problem(p):
    """p: dict with frequency/annoyance/willingness/searchability, each 0-10.
    Returns (total 0-100, verdict)."""
    for k in WEIGHTS:
        v = p.get(k)
        if v is None or not (0 <= v <= 10):
            raise ValueError(f"{k} must be 0-10, got {v!r}")
    total = round(sum(p[k] * w for k, w in WEIGHTS.items()) * 10, 1)
    if total >= 75:
        verdict = "做：高分痛点，优先上架"
    elif total >= 60:
        verdict = "可做：中等痛点，先做一版试水"
    else:
        verdict = "别做：痛点太弱，换题"
    return total, verdict



def _ascii(t):
    """Core fonts (helvetica) are latin-1 only: fold common Unicode
    punctuation to ASCII so a stray em-dash never kills a build."""
    return (
        str(t)
        .replace("\u2014", "-")
        .replace("\u2013", "-")
        .replace("\u201c", '"')
        .replace("\u201d", '"')
        .replace("\u2018", "'")
        .replace("\u2019", "'")
        .replace("\u2026", "...")
        .replace("\u00a0", " ")
    )


# ---------------------------------------------------------------- build
def build_pdf(spec, out_path):
    """spec: {title, problem, steps:[...], tips:[...], checklist:[...],
    price, author, cta_url}. Returns page count."""
    from fpdf import FPDF

    pdf = FPDF(format="Letter")
    pdf.set_auto_page_break(auto=True, margin=20)
    pdf.set_margins(20, 18, 20)

    # --- cover / title block
    pdf.add_page()
    pdf.set_font("helvetica", "B", 26)
    pdf.multi_cell(0, 11, _ascii(spec["title"]), new_x="LMARGIN", new_y="NEXT")
    pdf.ln(4)
    pdf.set_font("helvetica", "", 12)
    pdf.multi_cell(0, 6, _ascii("The problem: " + spec["problem"]), new_x="LMARGIN", new_y="NEXT")
    pdf.ln(6)
    pdf.set_font("helvetica", "I", 11)
    pdf.multi_cell(
        0, 6,
        "No theory. Just steps. Do it in about 10 minutes.",
        new_x="LMARGIN", new_y="NEXT",
    )
    pdf.ln(8)
    pdf.set_font("helvetica", "B", 13)
    pdf.cell(0, 8, _ascii("What you need"), new_x="LMARGIN", new_y="NEXT")
    pdf.set_font("helvetica", "", 11)
    for item in spec.get("needs", []):
        pdf.cell(5, 6, _ascii("-"), new_x="RIGHT", new_y="LAST")
        pdf.multi_cell(0, 6, _ascii(" " + item), new_x="LMARGIN", new_y="NEXT")

    # --- steps
    pdf.ln(4)
    pdf.set_font("helvetica", "B", 13)
    pdf.cell(0, 8, _ascii("Steps"), new_x="LMARGIN", new_y="NEXT")
    pdf.set_font("helvetica", "", 11)
    for i, step in enumerate(spec["steps"], 1):
        pdf.set_font("helvetica", "B", 11)
        pdf.cell(8, 6, _ascii(f"{i}."), new_x="RIGHT", new_y="LAST")
        pdf.set_font("helvetica", "", 11)
        pdf.multi_cell(0, 6, _ascii(" " + step), new_x="LMARGIN", new_y="NEXT")
        pdf.ln(1)

    # --- tips
    if spec.get("tips"):
        pdf.ln(2)
        pdf.set_font("helvetica", "B", 13)
        pdf.cell(0, 8, _ascii("Tips that save you a second try"), new_x="LMARGIN", new_y="NEXT")
        pdf.set_font("helvetica", "", 11)
        for tip in spec["tips"]:
            pdf.cell(5, 6, _ascii("-"), new_x="RIGHT", new_y="LAST")
            pdf.multi_cell(0, 6, _ascii(" " + tip), new_x="LMARGIN", new_y="NEXT")

    # --- checklist
    if spec.get("checklist"):
        pdf.ln(2)
        pdf.set_font("helvetica", "B", 13)
        pdf.cell(0, 8, _ascii("Checklist"), new_x="LMARGIN", new_y="NEXT")
        pdf.set_font("helvetica", "", 11)
        for item in spec["checklist"]:
            pdf.cell(5, 6, _ascii("[ ]"), new_x="RIGHT", new_y="LAST")
            pdf.multi_cell(0, 6, _ascii(" " + item), new_x="LMARGIN", new_y="NEXT")

    # --- footer
    pdf.ln(6)
    pdf.set_font("helvetica", "I", 10)
    pdf.multi_cell(
        0, 5,
        f"Made by {spec.get('author', 'Micro PDF Studio')}  |  "
        f"Price ${spec.get('price', 4.99):.2f}  |  "
        f"{spec.get('cta_url', '')}",
        new_x="LMARGIN", new_y="NEXT",
    )
    pdf.output(out_path)
    return pdf.pages_count


# ---------------------------------------------------------------- listing
def listing_pack(spec):
    """Gumroad 上架文案包 + 三条合规 BTW 留言模板（先帮忙、再顺带提链接）。"""
    title = spec["title"]
    price = spec.get("price", 4.99)
    desc = (
        f"{title}\n\n"
        f"Problem: {spec['problem']}\n\n"
        f"What you get: a short, no-theory PDF — {len(spec['steps'])} clear steps, "
        f"tips, and a checklist. Do it in about 10 minutes.\n\n"
        f"Price: ${price:.2f} — less than a latte."
    )
    btw = [
        "BTW, I had the same issue and wrote down the exact steps that fixed it for me: [link]",
        "I put together a quick PDF with the steps (took me 10 minutes to follow): [link] — hope it helps!",
        "Same problem here last week. I ended up writing a tiny guide for myself: [link]. No pressure, just sharing.",
    ]
    return {
        "gumroad_title": title,
        "price_usd": price,
        "description": desc,
        "btw_comments": btw,
        "anti_spam_rules": [
            "先给真实有用的回答，再顺带提链接；绝不在没帮忙的帖子下只丢链接。",
            "同一帖子只留一次链接；同一天不在同一社区发超过 3 条带链接的回复。",
            "链接文案写清是 PDF 指南 + 价格，不伪装成免费资源。",
        ],
    }


# ---------------------------------------------------------------- catalog
def catalog_add(path, slug, price, sales):
    data = {"products": []}
    if os.path.exists(path):
        with open(path) as f:
            data = json.load(f)
    for p in data["products"]:
        if p["slug"] == slug:
            p["sales"] += sales
            p["price"] = price
            break
    else:
        data["products"].append({"slug": slug, "price": price, "sales": sales})
    with open(path, "w") as f:
        json.dump(data, f, indent=2)
    return data


def catalog_total(path):
    with open(path) as f:
        data = json.load(f)
    total = sum(p["price"] * p["sales"] for p in data["products"])
    units = sum(p["sales"] for p in data["products"])
    return {"products": len(data["products"]), "units": units, "revenue_usd": round(total, 2)}


# ---------------------------------------------------------------- cli
def main(argv=None):
    ap = argparse.ArgumentParser(prog="micro_pdf.py")
    sub = ap.add_subparsers(dest="cmd", required=True)

    p = sub.add_parser("score")
    p.add_argument("--problem", required=True, help="JSON file with frequency/annoyance/willingness/searchability (0-10)")
    p = sub.add_parser("build")
    p.add_argument("--spec", required=True)
    p.add_argument("--out", required=True)
    p = sub.add_parser("listing")
    p.add_argument("--spec", required=True)
    p = sub.add_parser("catalog")
    p.add_argument("action", choices=["add", "total"])
    p.add_argument("--file", required=True)
    p.add_argument("--slug")
    p.add_argument("--price", type=float)
    p.add_argument("--sales", type=int, default=0)

    args = ap.parse_args(argv)
    if args.cmd == "score":
        with open(args.problem) as f:
            prob = json.load(f)
        total, verdict = score_problem(prob)
        print(json.dumps({"score": total, "verdict": verdict}, ensure_ascii=False))
    elif args.cmd == "build":
        with open(args.spec) as f:
            spec = json.load(f)
        pages = build_pdf(spec, args.out)
        print(json.dumps({"out": args.out, "pages": pages, "bytes": os.path.getsize(args.out)}))
    elif args.cmd == "listing":
        with open(args.spec) as f:
            spec = json.load(f)
        print(json.dumps(listing_pack(spec), indent=2, ensure_ascii=False))
    elif args.cmd == "catalog":
        if args.action == "add":
            data = catalog_add(args.file, args.slug, args.price, args.sales)
            print(json.dumps(data, indent=2))
        else:
            print(json.dumps(catalog_total(args.file), indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## 📈 模块五：安全合规、防封防踩坑与冷启闭环 SOP

### 平台政策红线

- **Reddit/社群分发**：只丢链接不帮忙 = 垃圾信息，会被删帖甚至封号。铁律：先给真实有用的回答，再顺带提链接；同一帖子只留一次链接；同一天同一社区不超 3 条带链接的回复；链接文案写清是 PDF 指南 + 价格，不伪装成免费资源。
- **Gumroad**：抽成约 10% 加支付手续费；买家退款会直接扣款，指南必须如标题所言、步骤真实可用，虚假承诺会招致退款与差评。
- **AI 生成内容**：指南内容须自己验证过步骤真实有效（咖啡机除味法须自己试过），不得编造不存在的功效；不得把 AI 生成的通用内容包装成“独家秘方”。

### 避坑清单

- 别只做新品不分发：作者原话“Stop creating. Start circulating.”——分发重于产量，旧产品重发、改写、换标题再分发。
- 别信截图里的“日均 $80–$400”：与 30 天总额对不上，未采用；只按 $1,171.92/30 天做测算。
- 别做合集：合集抬高了决策门槛，单品 3–19 美元的冲动价位才是这个模式成立的前提。
- 别碰医疗、法律、理财类“小麻烦”：专业建议有责任风险，只做生活类无风险小事（清洁、收纳、日常工具）。

### 冷启闭环（前 10 个种子付费用户）

1. 每个产品上架当天就去原帖留 BTW 回复（首单往往 24 小时内来）；
2. 把带来首单的帖子类型记下来（哪个社区、哪种问法），复制到同类帖子；
3. 首周目标不是收入数字，而是验证“有人愿为步骤付钱”——1 个付费用户即证明闭环成立。
